<script>

(function () {

  'use strict';


  const app = document.getElementById('app');

  let token = new URLSearchParams(location.search).get('token') || '';

  let activeTab = 'requests';


  function escapeHTML(str) {

    return String(str == null ? '' : str).replace(/[&<>"']/g, (m) => ({

      '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'

    }[m]));

  }


  function formatDate(iso) {

    try { return new Date(iso).toLocaleString(); } catch { return iso; }

  }


  async function apiFetch(path) {

    const res = await fetch(path, { headers: { 'x-admin-token': token } });

    const data = await res.json().catch(() => ({}));

    if (!res.ok) {

      throw new Error(data.error || 'Request failed (HTTP ' + res.status + ')');

    }

    return data;

  }


  function renderLogin(err) {

    app.innerHTML = `

      <div class="login">

        <h2>🔒 Admin Login</h2>

        <p class="sub">Enter your admin token to view requests</p>

        <input type="password" id="tokenInput" placeholder="Admin token" value="${escapeHTML(token)}" />

        <button id="loginBtn">Login</button>

        ${err ? '<p class="err">' + escapeHTML(err) + '</p>' : ''}

      </div>

    `;

    document.getElementById('loginBtn').addEventListener('click', () => {

      const v = document.getElementById('tokenInput').value.trim();

      if (!v) return renderLogin('Please enter a token.');

      token = v;

      history.replaceState(null, '', '?token=' + encodeURIComponent(v));

      load();

    });

    document.getElementById('tokenInput').addEventListener('keypress', (e) => {

      if (e.key === 'Enter') document.getElementById('loginBtn').click();

    });

  }


  async function load() {

    if (!token) return renderLogin();


    try {

      const statsData = await apiFetch('/api/admin/stats');

      let listData;

      if (activeTab === 'requests') {

        listData = await apiFetch('/api/admin/requests');

      } else {

        listData = await apiFetch('/api/admin/messages');

      }

      renderDashboard(statsData.stats, listData);

    } catch (err) {

      console.error(err);

      renderLogin(err.message || 'Invalid token or server error.');

    }

  }


  function renderDashboard(stats, listData) {

    const items = activeTab === 'requests' ? listData.requests : listData.messages;


    app.innerHTML = `

      <h1>📊 Shop Admin Panel</h1>

      <p class="sub">Manage customer requests and messages</p>


      <div class="stats">

        <div class="stat"><b>${stats.totalRequests}</b><span>Total Requests</span></div>

        <div class="stat"><b>${stats.newRequests}</b><span>New Requests</span></div>

        <div class="stat"><b>${stats.totalMessages}</b><span>Messages</span></div>

        <div class="stat"><b>${stats.totalProducts}</b><span>Products</span></div>

      </div>


      <div class="tabs">

        <button class="tab ${activeTab === 'requests' ? 'active' : ''}" data-tab="requests">

          🛒 Requests (${stats.totalRequests})

        </button>

        <button class="tab ${activeTab === 'messages' ? 'active' : ''}" data-tab="messages">

          📩 Messages (${stats.totalMessages})

        </button>

      </div>


      <div class="list" id="itemList"></div>

    `;


    app.querySelectorAll('.tab').forEach(btn => {

      btn.addEventListener('click', () => {

        activeTab = btn.dataset.tab;

        load();

      });

    });


    renderList(items);

  }


  function renderList(items) {

    const list = document.getElementById('itemList');

    if (!list) return;


    if (!items || items.length === 0) {

      list.innerHTML = '<div class="empty-msg">Nothing here yet. New submissions will appear here.</div>';

      return;

    }


    list.innerHTML = items.map(it => {

      if (activeTab === 'requests') {

        return `

          <div class="item">

            <div class="row">

              <div>

                <div class="name">${escapeHTML(it.item)}</div>

                <div class="meta">

                  <span><b>${escapeHTML(it.name)}</b></span>

                  <span>📞 ${escapeHTML(it.phone)}</span>

                  ${it.email ? '<span>✉️ ' + escapeHTML(it.email) + '</span>' : ''}

                  ${it.quantity ? '<span>📦 ' + escapeHTML(it.quantity) + '</span>' : ''}

                </div>

              </div>

              <div style="text-align:right">

                <span class="badge">${escapeHTML(it.status || 'new')}</span>

                <div class="time">${formatDate(it.createdAt)}</div>

              </div>

            </div>

            ${it.notes ? '<div class="notes">' + escapeHTML(it.notes) + '</div>' : ''}

          </div>

        `;

      } else {

        return `

          <div class="item">

            <div class="row">

              <div>

                <div class="name">${escapeHTML(it.name)}</div>

                <div class="meta">

                  <span>✉️ ${escapeHTML(it.email)}</span>

                  ${it.phone ? '<span>📞 ' + escapeHTML(it.phone) + '</span>' : ''}

                </div>

              </div>

              <div class="time">${formatDate(it.createdAt)}</div>

            </div>

            <div class="notes">${escapeHTML(it.message)}</div>

          </div>

        `;

      }

    }).join('');

  }


  load();

})();

</script>
